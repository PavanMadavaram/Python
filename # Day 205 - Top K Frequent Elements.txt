# Day 205 - Top K Frequent Elements
# DSA: Hash Map + Min-Heap

from collections import Counter
import heapq


def top_k_frequent(numbers, k):
    counts = Counter(numbers)

    heap = []
    for number, frequency in counts.items():
        heapq.heappush(heap, (frequency, number))

        if len(heap) > k:
            heapq.heappop(heap)

    return [number for frequency, number in heap]


numbers = [1, 1, 1, 2, 2, 3]
k = 2

print("Numbers:", numbers)
print(f"Top {k} frequent:", top_k_frequent(numbers, k))
