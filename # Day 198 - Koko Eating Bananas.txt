# Day 198 - Koko Eating Bananas
# DSA: Binary Search on the Answer

def hours_needed(piles, speed):
    return sum((pile + speed - 1) // speed for pile in piles)


def min_eating_speed(piles, hours):
    left = 1
    right = max(piles)

    while left < right:
        speed = (left + right) // 2

        if hours_needed(piles, speed) <= hours:
            right = speed
        else:
            left = speed + 1

    return left


piles = [3, 6, 7, 11]
hours = 8

speed = min_eating_speed(piles, hours)

print("Banana piles:", piles)
print("Available hours:", hours)
print("Minimum eating speed:", speed)
