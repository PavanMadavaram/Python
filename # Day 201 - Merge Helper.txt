# Day 201 - Merge Helper

def merge_sorted_arrays(first, second):
    result = []
    left = right = 0

    while left < len(first) and right < len(second):
        if first[left] <= second[right]:
            result.append(first[left])
            left += 1
        else:
            result.append(second[right])
            right += 1

    return result + first[left:] + second[right:]


print(merge_sorted_arrays([1, 3, 5], [2, 4, 6]))
print(merge_sorted_arrays([], [1, 2, 3]))
