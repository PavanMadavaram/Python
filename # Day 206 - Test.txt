import heapq


def kth_largest(numbers, k):
    heap = []

    for number in numbers:
        heapq.heappush(heap, number)

        if len(heap) > k:
            heapq.heappop(heap)

    return heap[0]


tests = [
    ([3, 2, 1, 5, 6, 4], 2, 5),
    ([3, 2, 3, 1, 2, 4, 5, 5, 6], 4, 4),
    ([1], 1, 1),
    ([10, 8, 6, 4], 3, 6),
]

for numbers, k, expected in tests:
    assert kth_largest(numbers, k) == expected
