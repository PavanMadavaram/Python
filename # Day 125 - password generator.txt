# Day 125 - password generator
import random
import string

def make_password(length=12):
    chars = string.ascii_letters + string.digits + "!@#$%"
    return "".join(random.choice(chars) for _ in range(length))

passwords = [make_password(10), make_password(12), make_password(14)]

for p in passwords:
    print(p)
