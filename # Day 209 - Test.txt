def min_subarray_length(target, numbers):
    left = 0
    current_sum = 0
    shortest = float("inf")

    for right, value in enumerate(numbers):
        current_sum += value

        while current_sum >= target:
            shortest = min(shortest, right - left + 1)
            current_sum -= numbers[left]
            left += 1

    return 0 if shortest == float("inf") else shortest


tests = [
    (7, [2, 3, 1, 2, 4, 3], 2),
    (4, [1, 4, 4], 1),
    (11, [1, 1, 1, 1], 0),
    (15, [1, 2, 3, 4, 5], 5),
]

for target, numbers, expected in tests:
    assert min_subarray_length(target, numbers) == expected
